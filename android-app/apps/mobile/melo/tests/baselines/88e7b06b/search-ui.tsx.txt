import React from 'react'
import {
	Pressable,
	ScrollView,
	StyleSheet,
	TextInput,
	View,
} from 'react-native'

import type { Creator } from './core/creators'
import type { SearchItem } from './core/model'
import { CreatorSearchResults } from './creator-remote-ui'
import type {
	CreatorSearchSession,
	CreatorSearchState,
} from './creator-search-session'
import { PageTitle } from './page-title'
import {
	musicResults,
	type SearchSession,
	type SearchState,
} from './search-session'
import { Button, Icon, State, T, shape, useTheme } from './ui'

export interface SearchPageProps {
	query: string
	filter: 'all' | 'music' | 'listening' | 'creator'
	search: SearchState & Pick<SearchSession, 'retry' | 'loadMore'>
	creatorSearch: { state: CreatorSearchState; session: CreatorSearchSession }
	onQueryChange: (query: string) => void
	onFilterChange: (filter: SearchPageProps['filter']) => void
	onSubmit: (value?: string) => void | Promise<void>
	onOpenCreator: (creator: Creator) => void
	renderRow: (item: SearchItem, index: number) => React.ReactNode
}

/** Presentation only. App retains both searches, their lifecycles and submission ownership. */
export function SearchPage({
	query,
	filter,
	search,
	creatorSearch,
	onQueryChange,
	onFilterChange,
	onSubmit,
	onOpenCreator,
	renderRow,
}: SearchPageProps) {
	const theme = useTheme()
	const c = theme.tokens
	const {
		submitted,
		items: results,
		ketangCount,
		more,
		loading: searching,
		error: searchError,
		failedPage,
		retry: retrySearch,
		loadMore,
	} = search
	const visible = filter === 'music' ? musicResults(results) : results
	return (
		<>
			<PageTitle
				title='寻找你的声音'
				subtitle='SEARCH / BILIBILI'
			/>
			<View
				style={[
					s.row,
					shape(theme.components.input),
					{
						backgroundColor: c.surface.card,
						borderColor: c.border.subtle,
						borderWidth: 1,
						paddingLeft: 13,
					},
				]}
			>
				<Icon
					name='search'
					color={c.text.secondary}
					size={19}
				/>
				<TextInput
					testID='search-input'
					accessibilityLabel='搜索关键词或 BV 号'
					value={query}
					onChangeText={onQueryChange}
					onSubmitEditing={(event) => {
						const value = event.nativeEvent.text
						onQueryChange(value)
						void onSubmit(value)
					}}
					returnKeyType='search'
					autoCapitalize='none'
					autoCorrect={false}
					placeholder={
						filter === 'creator'
							? 'UP 主昵称 / 关键词'
							: filter === 'listening'
								? '视频标题 / UP 主 / BV 号'
								: '歌曲关键词 / BV 号'
					}
					placeholderTextColor={c.text.secondary}
					style={{
						flex: 1,
						minHeight: 50,
						color: c.text.primary,
						paddingHorizontal: 10,
						fontSize: 14,
					}}
				/>
				<Pressable
					accessibilityRole='button'
					accessibilityLabel='提交搜索'
					onPress={() => {
						void onSubmit()
					}}
					style={{
						minWidth: 58,
						minHeight: 48,
						alignItems: 'center',
						justifyContent: 'center',
					}}
				>
					<T
						style={{ color: c.action.primary }}
						weight='600'
					>
						搜索
					</T>
				</Pressable>
			</View>
			<ScrollView
				horizontal
				showsHorizontalScrollIndicator={false}
				contentContainerStyle={[s.row, { gap: 12 }]}
				style={{ marginVertical: 15 }}
			>
				{(['all', 'music', 'listening', 'creator'] as const).map((value) => (
					<Pressable
						key={value}
						accessibilityRole='tab'
						accessibilityState={{ selected: filter === value }}
						onPress={() => onFilterChange(value)}
						style={{
							minHeight: 48,
							paddingHorizontal: 14,
							justifyContent: 'center',
							borderBottomWidth: filter === value ? 2 : 0,
							borderBottomColor: c.action.primary,
						}}
					>
						<T
							size={13}
							weight={filter === value ? '700' : '400'}
						>
							{value === 'all'
								? '全部音源'
								: value === 'music'
									? '音乐分区'
									: value === 'listening'
										? '听视频'
										: 'UP 主'}
						</T>
					</Pressable>
				))}
			</ScrollView>
			{filter === 'creator' ? (
				<CreatorSearchResults
					state={creatorSearch.state}
					session={creatorSearch.session}
					onOpen={onOpenCreator}
				/>
			) : (
				<>
					{' '}
					{submitted && (
						<T
							muted
							size={11}
						>
							“{submitted}” · {results.length} 条已加载 · {visible.length}{' '}
							条可见 ·{' '}
							{filter === 'listening' ? '保留原标题与分 P' : '歌曲版本待核对'}
						</T>
					)}
					{!submitted && !searching && !searchError ? (
						<State
							title={
								filter === 'listening'
									? '想听的内容，从这里找到'
									: '喜欢的歌，从这里遇见'
							}
							message={
								filter === 'listening'
									? '输入视频原标题、UP 主关键词或完整 BV 号。无需登录 B 站。'
									: '输入歌曲、UP 主关键词或完整 BV 号。无需登录 B 站。'
							}
						/>
					) : searchError && failedPage === 1 ? (
						<State
							title='搜索未完成'
							message={searchError}
							retry={() => {
								void retrySearch()
							}}
						/>
					) : searching && results.length === 0 ? (
						<State
							busy
							title='正在寻找真实音源'
							message='从 B 站读取标题、封面和上传者信息'
						/>
					) : visible.length === 0 ? (
						<State
							title={
								results.length > 0
									? '当前筛选没有匹配音源'
									: ketangCount > 0
										? '当前结果没有可展示的视频'
										: '没有找到结果'
							}
							message={
								filter === 'music' && results.length > 0
									? more
										? '已加载的音源未标为音乐分区，可查看全部音源，或继续加载下一页。'
										: '已加载的音源未标为音乐分区，可查看全部音源或换一个词。'
									: results.length === 0 && ketangCount > 0
										? more
											? '已读取的结果仅包含暂不支持的 B 站课程，可继续加载下一页。'
											: '已读取的结果仅包含暂不支持的 B 站课程，可更换关键词或使用完整 BV 号。'
										: '试试更短的关键词，或使用完整 BV 号。'
							}
						/>
					) : (
						<>{visible.map(renderRow)}</>
					)}
					{filter === 'music' && results.length > 0 && visible.length === 0 && (
						<Button
							secondary
							onPress={() => onFilterChange('all')}
						>
							查看全部音源（{results.length}）
						</Button>
					)}
					{!!searchError && failedPage !== 1 && (
						<State
							title='下一页加载失败'
							message={searchError}
							retry={() => {
								void retrySearch()
							}}
						/>
					)}
					{more && !searchError && (
						<Button
							secondary
							disabled={searching}
							onPress={() => {
								void loadMore()
							}}
						>
							{searching ? '正在加载…' : '加载更多结果'}
						</Button>
					)}{' '}
				</>
			)}
		</>
	)
}
const s = StyleSheet.create({
	row: { flexDirection: 'row', alignItems: 'center' },
})
